var train = header.Find("TrainImage");
var cars = train.Cast<UnityEngine.Transform>().Where(t => t.name.StartsWith("CarSlot_")).ToArray();
if (cars.Length != 4) return "Expected exactly four existing car slots";
UnityEditor.Undo.RegisterFullObjectHierarchyUndo(header.gameObject, "Style ark train header");
UnityEditor.AssetDatabase.Refresh();
string root = "Assets/Sprites/UI/Train/";
foreach (string file in new [] { "ArkLocomotive.png", "ArkCarriage.png", "ArkExpansionCar.png" }) {
    var importer = (UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(root + file);
    importer.textureType = UnityEditor.TextureImporterType.Sprite;
    importer.spriteImportMode = UnityEditor.SpriteImportMode.Single;
    importer.spritePixelsPerUnit = 100;
    importer.alphaIsTransparency = true;
    importer.mipmapEnabled = false;
    importer.maxTextureSize = 2048;
    importer.textureCompression = UnityEditor.TextureImporterCompression.Uncompressed;
    importer.SaveAndReimport();
}
System.Action<UnityEngine.RectTransform,float,float,float,float> anchor = (r,x0,y0,x1,y1) => {
    r.anchorMin = new UnityEngine.Vector2(x0,y0); r.anchorMax = new UnityEngine.Vector2(x1,y1);
    r.pivot = new UnityEngine.Vector2(.5f,.5f); r.offsetMin = UnityEngine.Vector2.zero; r.offsetMax = UnityEngine.Vector2.zero;
};
System.Func<UnityEngine.Transform,string,UnityEngine.GameObject> child = (parent,name) => {
    var existing = parent.Find(name); if (existing != null) return existing.gameObject;
    var go = new UnityEngine.GameObject(name,typeof(UnityEngine.RectTransform));
    UnityEditor.Undo.RegisterCreatedObjectUndo(go,"Create train decoration"); go.transform.SetParent(parent,false); return go;
};
System.Func<UnityEngine.Transform,string,string,float,float,float,float,float,UnityEngine.Color,TMPro.TextMeshProUGUI> label = (parent,name,text,x0,y0,x1,y1,size,color) => {
    var go=child(parent,name); var tmp=go.GetComponent<TMPro.TextMeshProUGUI>(); if(tmp==null) tmp=go.AddComponent<TMPro.TextMeshProUGUI>();
    anchor((UnityEngine.RectTransform)go.transform,x0,y0,x1,y1); tmp.text=text; tmp.font=menu.mainFont;
    tmp.fontSize=size; tmp.color=color; tmp.alignment=TMPro.TextAlignmentOptions.Center; tmp.raycastTarget=false;
    tmp.enableWordWrapping=false; tmp.overflowMode=TMPro.TextOverflowModes.Ellipsis; return tmp;
};
